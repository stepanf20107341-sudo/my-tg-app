import asyncio import json from aiogram import Bot, Dispatcher, types from aiogram.filters import CommandStart from aiogram.types import WebAppInfo, MenuButtonWebApp # !!! СЮДИ ВСТАВ СВІЙ ТОКЕН ВІД BOTFATHER МІЖ КАВИЧКАМИ !!! BOT_TOKEN = "СЮДИ_ВСТАВ_СВІЙ_ТОКЕН" # Твоє постійне та перевірене посилання з GitHub Pages WEB_APP_URL = "https://github.io" bot = Bot(token=BOT_TOKEN) dp = Dispatcher() # Обробка команди /start @dp.message(CommandStart()) async def cmd_start(message: types.Message): # Налаштовуємо офіційну кнопку "Кабінет ⚡" ліворуч від поля введення тексту await bot.set_chat_menu_button( chat_id=message.chat.id, menu_button=MenuButtonWebApp( text="Кабінет ⚡", web_app=WebAppInfo(url=WEB_APP_URL) ) ) await message.answer( f"Вітаємо у каталозі Вірт-Моделей України, {message.from_user.first_name}!\n\n" f"Внизу екрана (ліворуч від поля введення) з'явилася кнопка «Кабінет ⚡».\n" f"Натисни її, щоб переглянути каталог моделей або створити власну анкету!" ) # Бот автоматично ловить сповіщення, коли модель створює нову анкету через сайт @dp.message(lambda message: message.web_app_data is not None) async def handle_web_app_data(message: types.Message): try: data = json.loads(message.web_app_data.data) if data.get("action") == "new_model": text_response = ( f"👑 **Зареєстровано нову модель в агентстві!**\n\n" f"👤 Ім'я дівчини: {data.get('name')}\n" f"💰 Початкова ціна (вірт): {data.get('price')} грн\n" f"💬 Telegram для зв'язку: @{message.from_user.username if message.from_user.username else 'Немає юзернейму'}" ) await message.answer(text_response, parse_mode="Markdown") except Exception: pass async def main(): print("🚀 Бот Агентства Вірт-Моделей успішно запущений!") print("Поточне посилання додатка: " + WEB_APP_URL) await dp.start_polling(bot) if __name__ == "__main__": asyncio.run(main())
